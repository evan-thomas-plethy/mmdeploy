python tools/deploy.py configs/mmpose/pose-detection_simcc_coreml_static-256x192.py mmpose/configs/rtmpose-m_8xb256-420e_aic-coco-256x192.py mmpose/checkpoints/rtmpose-m_pretrained.pth demo.jpg --work-dir work_dir/rtmpose-m_pretrained --device cpu --dump-info
